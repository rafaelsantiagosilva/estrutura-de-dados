Listas Ligadas ou **Listas Encadeadas** (nome que faz muito mais sentido, e já digo o porquê) são estruturas de dados em que elementos são formados por `structs` que contém um **valor** (ou mais) e uma **referência** para outro elemento. 

> Chamamos esses elementos de **nós** (ou **nodos**)

Basicamente essas listas funcionam como correntes. Cada elemento é encadeado em outro, que por sua vez é encadeado em outro...

Abaixo temos a estrutura básica de uma lista.

```c
#include<stdio.h>
#include<stdlib.h>

typedef struct No {
    int valor;
    struct No *prox;
} No;

typedef struct Lista {
    struct No *ini;
} Lista;
``` 

A estrutura `No` possui uma referência para uma outra estrutura `No`, através do ponteiro (ou apontador) `prox`.

## Funções

Para simplificar processos repetitivos, podemos criar funções para manipular melhor as listas.

### `newNo`

```c
// construtor do nó - cria um novo nó na memória
No *newNo(int valor)
{
    No *no = (No*)malloc(sizeof(No));
    no->valor = valor;
    no->prox = NULL; // não temos um outro valor
    return no;
}
```

### `newLista`

```c
// construtor da lista - cria uma nova lista na memória
Lista *newLista()
{
    Lista *li = (Lista*)malloc(sizeof(Lista));
    li->ini = NULL; // começa vazia
    return li;
}
```

### `estaVazia`

```c
// retorna 1 se está vazia - nenhum elemento foi adicionado
int estaVazia(Lista *li)
{
    // tá sem nó no início?
    return li->ini == NULL;
}
```

### `size`

```c
// conta até achar o nó final (nó que aponta para NULL)
int size(Lista *li)
{
    int size = 0;

    // nó auxiliar: vamos percorrer a lista usando ele
    No *aux = li->ini;

    while (aux != NULL)
    {
        aux = aux->prox;
        size++;
    } // quando chegar em NULL acaba

    return size;
}
```

### `printLista`

```c
// imprime a lista
void printLista(Lista *li)
{
    printf("[");

    // nó auxiliar: vamos percorrer a lista usando ele
    No *aux = li->ini;
    
    while(aux != NULL) 
    {
        printf("%d", aux->valor);

        aux = aux->prox;

        if (aux != NULL)
        {
            printf(", ");
        }
    }

    printf("]");

    // imprime o tamanho da lista
    printf("(%d)\n", size(li));
}
```

### `addFim`

```c
void addFim(Lista *li, int valor)
{
    No *no = newNo(valor);

    // lista vazia?
    if(li->ini == NULL)
    {
        li->ini = no; // começa a lista
    } 
    else
    {
        No *aux;
        aux = li->ini;

        // até chegar no último nó
        while(aux->prox!=NULL)
        {
            aux = aux->prox;
        }

        // aux agora é o último nó
        aux->prox = no;
    }
}
```

### `addInicio`

```c
// insere o valor na primeira posição
void addInicio(Lista *li, int valor)
{
    No *no = newNo(valor);

    no->prox = li->ini; // o novo nó aponta para quem era o primeiro
    li->ini = no;       // o novo nó vira o primeiro
}
```

### `get`

```c
// devolve um apontador para o nó na posição idx (começa em 0)
No *get(Lista *li, int idx)
{
    No *aux = li->ini;

    // anda idx vezes
    for (int i = 0; i < idx; i++)
    {
        if (aux != NULL)
        {
            aux = aux->prox;
        }
    }

    // se saiu da lista, o índice era inválido
    if (aux == NULL)
    {
        printf("Fora dos limites\n");
        exit(1);
    }

    return aux;
}
```

### `set`

```c
// substitui o valor da posição idx pelo parâmetro 'valor'
void set(Lista *li, int valor, int idx)
{
    No *aux = get(li, idx); // acha o nó (custo O(n))
    aux->valor = valor;     // troca o valor (custo Θ(1))
}
```

### `remover`

```c
// remove o elemento da posição idx e devolve o valor removido
int remover(Lista *li, int idx)
{
    No *no; // nó que será removido

    if (idx == 0)
    {
        no = li->ini;
        li->ini = li->ini->prox; // o segundo vira o primeiro
    }
    else
    {
        No *anterior = get(li, idx - 1); // custo O(n)
        no = anterior->prox;
        anterior->prox = anterior->prox->prox; // pula o nó
    }

    int valor = no->valor;
    free(no); // devolve a memória (evita vazamento!)
    return valor;
}
```

### `add`

```c
// insere o valor na posição idx; os demais "andam" para a direita
void add(Lista *li, int valor, int idx)
{
    No *no = newNo(valor);

    if (idx == 0)
    {
        no->prox = li->ini;
        li->ini = no;
    }
    else
    {
        No *anterior = get(li, idx - 1);
        no->prox = anterior->prox; // 1º: o novo nó segura o resto da lista
        anterior->prox = no;       // 2º: o anterior passa a apontar para o novo
    }
}
```

### `busca`

```c
// devolve 1 se k pertence à lista, 0 caso contrário
int busca(Lista *li, int k)
{
    No *aux = li->ini;

    while (aux != NULL)
    {
        if (aux->valor == k)
        {
            return 1; // achou
        }
        aux = aux->prox; // próximo nó, sem recomeçar do zero
    }

    return 0; // chegou no fim sem achar
}
```

### `addOrdenado`

```c
// insere o valor na lista de maneira ordenada (crescente)
void addOrdenado(Lista *li, int valor)
{
    No *no = newNo(valor);

    // lista vazia OU valor é menor/igual ao primeiro: entra no começo
    if (li->ini == NULL || valor <= li->ini->valor)
    {
        no->prox = li->ini;
        li->ini = no;
    }
    else
    {
        No *aux = li->ini;

        // para no último nó cujo PRÓXIMO ainda é menor que valor
        while (aux->prox != NULL && aux->prox->valor < valor)
        {
            aux = aux->prox;
        }

        no->prox = aux->prox; // 1º: segura o resto
        aux->prox = no;       // 2º: encaixa
    }
}
```
